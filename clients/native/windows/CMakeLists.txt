cmake_minimum_required(VERSION 3.22)
project(duct_native LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(json)
add_library(duct_native_plugin SHARED plugin/duct_native_plugin.cpp plugin/managed_tunnel.cpp plugin/connection_check.cpp)
target_link_libraries(duct_native_plugin PRIVATE flutter flutter_wrapper_plugin nlohmann_json::nlohmann_json advapi32 crypt32 shell32 ole32 iphlpapi ws2_32 dnsapi winhttp)
target_include_directories(duct_native_plugin INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_definitions(duct_native_plugin PRIVATE DUCT_NATIVE_PLUGIN_IMPL)
set(duct_native_bundled_libraries "${CMAKE_CURRENT_SOURCE_DIR}/bin/duct_bridge.dll" PARENT_SCOPE)
add_executable(dtvpn-agent service/agent.cpp service/filter.cpp)
target_link_libraries(dtvpn-agent PRIVATE fwpuclnt advapi32 rpcrt4 ws2_32)
